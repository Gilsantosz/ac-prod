-- Execuções limpas permanecem como auditoria, mas deixam de reservar o
-- arquivo/prefixo sintético. A constraint original era global e contradizia o
-- trigger prefix-free, que já ignora status cleaned por projeto e ambiente.

begin;

alter table public.capacity_test_runs
  drop constraint if exists capacity_test_runs_synthetic_prefix_key;

create unique index if not exists uq_capacity_test_runs_live_prefix
  on public.capacity_test_runs(project_ref, environment, synthetic_prefix)
  where status <> 'cleaned';

comment on index public.uq_capacity_test_runs_live_prefix is
  'Reserva prefixos apenas para execuções não limpas; permite repetir com segurança o mesmo arquivo depois da limpeza transacional.';
